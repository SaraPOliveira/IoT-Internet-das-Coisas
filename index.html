📌 Sobre o Projeto

Projeto IoT utilizando Arduino com Ethernet Shield para comunicação em rede local, funcionando como servidor web e permitindo controle de dispositivos físicos via navegador.

🎯 Objetivos

  • Conectar Arduino à rede
  • Configurar IP fixo no Arduino
  • Criar servidor HTTP
  • Controlar LEDs remotamente
  • Testar conectividade via Ping

🔧 Componentes

  • Arduino UNO
  • Ethernet Shield (W5100)
  • Roteador
  • 2 Cabos Ethernet
  • Smartphone com app de Ping

💻 Código Arduino

#include <SPI.h>
#include <Ethernet.h>

#define led1 8
#define led2 9

byte mac[] = { 0x90, 0xA2, 0xDA, 0xB6, 0x25, 0x3A };
EthernetServer server(80);

void setup() {
  pinMode(led1, OUTPUT);
  pinMode(led2, OUTPUT);

  Serial.begin(9600);

  IPAddress ip(192,168,0,122);
  Ethernet.begin(mac, ip);
  server.begin();

  Serial.print("IP: ");
  Serial.println(Ethernet.localIP());
}

void loop() {
  EthernetClient client = server.available();

  if (client) {
    String request = "";

    while (client.connected()) {
      if (client.available()) {
        char c = client.read();
        request += c;
        if (c == '\n') break;
      }
    }

    if (request.indexOf("led1-on") >= 0) digitalWrite(led1, HIGH);
    if (request.indexOf("led1-off") >= 0) digitalWrite(led1, LOW);
    if (request.indexOf("led2-on") >= 0) digitalWrite(led2, HIGH);
    if (request.indexOf("led2-off") >= 0) digitalWrite(led2, LOW);

    client.println("HTTP/1.1 200 OK");
    client.println("Content-Type: text/html");
    client.println();
    client.println("<h1>Arduino IoT</h1>");

    client.stop();
  }
}
